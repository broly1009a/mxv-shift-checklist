Sub Tao_file_ban_tin()
    Dim FSO As Object
         Set FSO = CreateObject("Scripting.FileSystemObject")
    Dim twb, nwb, nwb1 As Workbook
        Set twb = ThisWorkbook
    Dim sour, des, fd As Range
        Set sour = twb.Sheets("sheet2").Range("A6")
        Set des = twb.Sheets("sheet2").Range("A7")
        Set fd = twb.Sheets("sheet2").Range("A4")
    If FSO.FolderExists(fd) = False Then
    MsgBox "khong ton tai thu muc nguon"
    Exit Sub
    End If
    FSO.CopyFile sour, des
    Set FSO = Nothing
    Set nwb = Workbooks.Open(des)
    twb.Sheets(1).Range("L11:L77").Copy
    nwb.ActiveSheet.Range("D6").PasteSpecial xlPasteValues
    twb.Sheets(1).Range("T11:T13").Copy
    nwb.ActiveSheet.Range("D73").PasteSpecial xlPasteValues
    nwb.ActiveSheet.Columns("D:D").NumberFormat = "#,##0 "
    
    Set FSO = CreateObject("Scripting.FileSystemObject")
    Set sour = twb.Sheets("sheet2").Range("A6")
        Set des = twb.Sheets("sheet2").Range("A26")
        Set fd = twb.Sheets("sheet2").Range("A24")
    If FSO.FolderExists(fd) = False Then
    MsgBox "khong ton tai thu muc nguon"
    Exit Sub
    End If
    FSO.CopyFile sour, des
    Set FSO = Nothing
    Set nwb1 = Workbooks.Open(des)
    twb.Sheets(1).Range("L11:L77").Copy
    nwb1.ActiveSheet.Range("D6").PasteSpecial xlPasteValues
    twb.Sheets(1).Range("T11:T13").Copy
    nwb1.ActiveSheet.Range("D73").PasteSpecial xlPasteValues
    nwb1.Save
    nwb1.Close False
    
End Sub
